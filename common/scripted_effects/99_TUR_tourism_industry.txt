TUR_startup = {
	set_variable = { TUR_tourism_quantity = 0 }
	set_variable = { TUR_tourism_quality = 0 }

TUR_the_first_party = {
	clamp_variable = {
		var = TUR_tourism_quantity
		min = 0
		max = 100
	}
}
TUR_the_second_party = {
	clamp_variable = {
		var = TUR_tourism_quality
		min = 0
		max = 100
	}
}
